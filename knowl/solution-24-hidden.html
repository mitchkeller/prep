(a) Between \((3, 34)\) and \((5, 82)\text{,}\)
\begin{equation*} \hbox{average rate of change }=\frac{\Delta y}{\Delta x}= \frac{82 - 34}{5 - 3}= \frac{48}{2} = 24 . \end{equation*}
(b) Between \((c, k)\) and \((q, t)\text{,}\)
\begin{equation*} \hbox{average rate of change }=\frac{\Delta y}{\Delta x}= \frac{t - k}{q - c}. \end{equation*}
(c) Between \((x,\ f(x))\) and \((x+h,\ f(x+h))\text{,}\)
\begin{equation*} \begin{aligned} \hbox{average rate of change } \amp =\frac{\Delta y}{\Delta x} = \frac{f(x+h)-f(x)}{(x+h)-x}\\ \amp = \frac{3\!\left(x+h\right)^{2}+7-\left(3x^{2}+7\right)}{x+h-x} \\ \amp =\frac{3 x^2 + 6 x h + 3 h^2 + 7 - (3 x^2 + 7)}{h}\\ \amp = \frac{6 x h + 3 h^2}{h}\\ \amp = 6x+3h. \end{aligned} \end{equation*}
